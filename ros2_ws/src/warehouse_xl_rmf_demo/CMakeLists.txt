cmake_minimum_required(VERSION 3.8)
project(warehouse_xl_rmf_demo)

find_package(ament_cmake REQUIRED)
# plansys2_executor's exported targets pull in Eigen3 without finding it, so a
# consumer has to. Without this the configure step fails on a target that is
# nowhere in this file.
find_package(Eigen3 REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plansys2_msgs REQUIRED)
find_package(plansys2_domain_expert REQUIRED)
find_package(plansys2_executor REQUIRED)
find_package(plansys2_planner REQUIRED)
find_package(plansys2_problem_expert REQUIRED)

set(dependencies
  lifecycle_msgs
  rclcpp
  plansys2_msgs
  plansys2_domain_expert
  plansys2_executor
  plansys2_planner
  plansys2_problem_expert
)

# The multi-site mission. The single-site one is eplansys_demo's and is used
# unchanged; this one exists because what has to be added to the problem is
# different -- three sites as well as three agents -- and because the policy it
# gets back is worth printing as the tree it is.
add_executable(survey_sites_mission src/survey_sites_mission.cpp)
ament_target_dependencies(survey_sites_mission ${dependencies})

install(TARGETS survey_sites_mission
  DESTINATION lib/${PROJECT_NAME}
)

# The chase camera talks to Gazebo and to nothing else, so it is built against
# gazebo's own transport rather than against gazebo_ros. It is a demonstration
# aid: the mission does not depend on it and nothing depends on it running.
find_package(gazebo REQUIRED)
add_executable(chase_camera src/chase_camera.cpp)
target_include_directories(chase_camera PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(chase_camera PRIVATE ${GAZEBO_LIBRARY_DIRS})
# tbb explicitly. Gazebo's exported link line pulls libtbb in transitively
# through ignition-transport but does not name it, so the linker sees the
# symbols and not the library.
target_link_libraries(chase_camera ${GAZEBO_LIBRARIES} tbb)

install(TARGETS chase_camera
  DESTINATION lib/${PROJECT_NAME}
)

# worlds, maps and models are generated by tools/ and installed as data. The
# generators are installed too, because the floor is meant to be regenerated
# rather than edited: every number in the world and the graph comes from the
# world's own collision meshes.
install(PROGRAMS scripts/scan_perception.py scripts/site_perception.py
                scripts/record_models.py
  DESTINATION lib/${PROJECT_NAME}
)

install(DIRECTORY launch config maps worlds models tools docs epddl pddl params
  DESTINATION share/${PROJECT_NAME}
)

ament_package()
